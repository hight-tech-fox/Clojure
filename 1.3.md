Определить функции my-map и my-filter, аналогичные map (для одного списка) и filter, выразив их через reduce и базовые операции над списками (cons, first, concat и т.п.).

(defn my-map [f items]
  (reduce
    (fn [result item]
      (concat result (cons (f item) '())))
    '()
    items))


(defn my-filter [predicate items]
  (reduce
    (fn [result item]
      (if (predicate item)
        (concat result (cons item '()))
        result))
    '()
    items))


(defn add-ten [num]
  (+ num 10))


(prn (my-map add-ten '(1 2 3)))
