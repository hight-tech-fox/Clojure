# **Task 1.1.**



Задан набор (алфавит) символов в виде списка и число n. Опишите функцию, которая возвращает список всех строк длины n, состоящих из этих символов и не содержащих двух одинаковых символов, идущих подряд.



Для решения использовать рекурсию и базовые операции над константами и списками (str, cons, .concat и т.п.)



Пример: для алфавита ("a" "b" "c") и n=2 результат должен быть ("ab" "ac" "ba" "bc" "ca" "cb") с точностью до перестановки.

(defn apply-n-times \[n f x]

&#x20; (if (zero? n)

&#x20;   x

&#x20;   (apply-n-times (dec n) f (f x))))



(defn append-char-to-all \[words ch]

&#x20; (if (empty? words)

&#x20;   '()

&#x20;   (cons (str (first words) ch)

&#x20;         (append-char-to-all (rest words) ch))))



(defn append-alphabet-to-all \[alphabet words]

&#x20; (if (empty? alphabet)

&#x20;   '()

&#x20;   (concat

&#x20;     (append-char-to-all words (first alphabet))

&#x20;     (append-alphabet-to-all (rest alphabet) words))))



(defn valid-word? \[chars]

&#x20; (if (empty? (rest chars))

&#x20;   true

&#x20;   (if (= (first chars) (first (rest chars)))

&#x20;     false

&#x20;     (valid-word? (rest chars)))))



(defn keep-valid-words \[words]

&#x20; (if (empty? words)

&#x20;   '()

&#x20;   (if (valid-word? (first words))

&#x20;     (cons (first words)

&#x20;           (keep-valid-words (rest words)))

&#x20;     (keep-valid-words (rest words)))))



(defn generate-words \[alphabet n]

&#x20; (apply-n-times

&#x20;   n

&#x20;   (fn \[words]

&#x20;     (keep-valid-words

&#x20;       (append-alphabet-to-all alphabet words)))

&#x20;   '("")))
