(ns main)

(defn split-blocks [items size]
  (partition-all size items))

(defn parallel-filter [predicate items size]
  (letfn [(process-groups [groups]
            (lazy-seq
              (when-let [remaining (seq groups)]
                (let [tasks
                      (mapv
                        (fn [block]
                          (future
                            (vec (filter predicate block))))
                        (first remaining))]
                  (concat
                    (mapcat deref tasks)
                    (process-groups (rest remaining)))))))]
    (process-groups
      (partition-all 4 (split-blocks items size)))))

(defn slow-even? [x]
  (Thread/sleep 2)
  (even? x))

(def numbers (range 200))

(println "Standard filter:")
(time (vec (filter slow-even? numbers)))

(println "Parallel filter:")
(time (vec (parallel-filter slow-even? numbers 20)))

(println "Results are equal:")
(println
  (= (vec (filter even? numbers))
     (vec (parallel-filter even? numbers 20))))

;; Laziness test

(def checked (atom 0))

(defn counted-even? [x]
  (swap! checked inc)
  (even? x))

(def lazy-result
  (parallel-filter counted-even? (range 200) 20))

(println "Checked before requesting elements:" @checked)
(assert (zero? @checked))

(println "First result:" (first lazy-result))
(println "Checked after requesting first element:" @checked)
(assert (= @checked 80))

(println "Total results:" (count lazy-result))
(println "Checked after requesting all elements:" @checked)
(assert (= @checked 200))

(println "Laziness test passed.")
