(ns main
  (:require [clojure.test :refer [deftest is run-tests]]))

(defmulti evaluate
  (fn [expr values]
    (cond
      (vector? expr) (first expr)
      (symbol? expr) :variable
      :else :constant)))

(defmethod evaluate :constant [expr values]
  expr)

(defmethod evaluate :variable [expr values]
  (get values expr))

(defmethod evaluate :and [[_ & args] values]
  (every? #(evaluate % values) args))

(defmethod evaluate :or [[_ & args] values]
  (boolean (some #(evaluate % values) args)))

(defmethod evaluate :not [[_ a] values]
  (not (evaluate a values)))

(defmethod evaluate :implies [[_ a b] values]
  (or (not (evaluate a values))
      (evaluate b values)))

(defmethod evaluate :xor [[_ a b] values]
  (not= (evaluate a values)
        (evaluate b values)))

(defn variables [expr]
  (cond
    (symbol? expr) [expr]
    (vector? expr) (mapcat variables (rest expr))
    :else []))

(defn assignments [vars]
  (reduce
    (fn [rows variable]
      (for [row rows
            value [false true]]
        (assoc row variable value)))
    [{}]
    vars))

(defn make-term [vars row]
  (let [items (mapv (fn [v]
                     (if (get row v)
                       v
                       [:not v]))
                   vars)]
    (if (= (count items) 1)
      (first items)
      (into [:and] items))))


(defn dnf
  "Converts a boolean expression to DNF."
  [expr]
  (let [vars (sort (distinct (variables expr)))
        rows (assignments vars)
        valid (filter #(evaluate expr %) rows)]
    (cond
      (empty? valid) false
      (= (count valid) (count rows)) true
      (= (count valid) 1) (make-term vars (first valid))
      :else (into [:or] (map #(make-term vars %) valid)))))

(defn replace-var [expr variable value]
  (cond
    (= expr variable) value
    (vector? expr)
    (mapv #(replace-var % variable value) expr)
    :else expr))

(defn substitute
  "Replaces a variable and converts the result to DNF."
  [expr variable value]
  (dnf (replace-var expr variable value)))

(deftest boolean-tests
  (is (= true
         (evaluate [:and 'x 'y] {'x true 'y true})))

  (is (= false
         (evaluate [:or 'x 'y] {'x false 'y false})))

  (is (= false
         (evaluate [:not 'x] {'x true})))

  (is (= false
         (evaluate [:implies 'x 'y] {'x true 'y false})))

  (is (= true
         (evaluate [:xor 'x 'y] {'x true 'y false})))

  (is (= [:and 'x 'y]
         (dnf [:and 'x 'y])))

  (is (= true
         (dnf [:or 'x [:not 'x]])))

  (is (= 'y
         (substitute [:and 'x 'y] 'x true)))

  (is (= false
         (substitute [:and 'x 'y] 'x false))))


(println (dnf [:implies 'x 'y]))
(println (substitute [:and 'x 'y] 'x true))

(run-tests)
