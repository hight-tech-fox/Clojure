(defn apply-n-times [n f x]
  (loop [remaining n
         result x]
    (if (zero? remaining)
      result
      (recur (dec remaining) (f result)))))


(defn append-char-to-all [words ch]
  (loop [remaining words
         result '()]
    (if (empty? remaining)
      (reverse result)
      (recur (rest remaining)
             (cons (str (first remaining) ch)
                   result)))))


(defn append-alphabet-to-all [alphabet words]
  (loop [remaining alphabet
         result '()]
    (if (empty? remaining)
      result
      (recur (rest remaining)
             (concat result
                     (append-char-to-all
                       words
                       (first remaining)))))))


(defn valid-word? [chars]
  (loop [remaining (seq chars)]
    (if (empty? (rest remaining))
      true
      (if (= (first remaining)
             (first (rest remaining)))
        false
        (recur (rest remaining))))))


(defn keep-valid-words [words]
  (loop [remaining words
         result '()]
    (if (empty? remaining)
      (reverse result)
      (recur (rest remaining)
             (if (valid-word? (first remaining))
               (cons (first remaining) result)
               result)))))


(defn generate-words [alphabet n]
  (apply-n-times
    n
    (fn [words]
      (keep-valid-words
        (append-alphabet-to-all alphabet words)))
    '("")))
